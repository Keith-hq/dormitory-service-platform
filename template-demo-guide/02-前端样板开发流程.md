## 二、分步操作

### 2.1 创建 Vite + Vue3 项目

```bash
npm create vite@latest template-frontend -- --template vue
cd template-frontend
npm install
```

### 2.2 安装核心依赖

```bash
npm install pinia vue-router axios
```

## 三、代码模板

### 3.1 请求封装 `src/utils/request.js`

```js
import axios from 'axios'
import { useUserStore } from '@/store/user'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000
})

// 请求拦截器：携带 JWT Token
request.interceptors.request.use(config => {
  const userStore = useUserStore()
  if (userStore.token) {
    config.headers.Authorization = `Bearer ${userStore.token}`
  }
  return config
})

// 响应拦截器：统一错误处理
request.interceptors.response.use(
  response => {
    const { code, message, data } = response.data
    if (code === 200) return data
    return Promise.reject(new Error(message || '请求失败'))
  },
  error => Promise.reject(error)
)

export default request
```

### 3.2 API 接口封装 `src/api/building.js`

```js
import request from '@/utils/request'

export const buildingApi = {
  getList: (params) => request.get('/building', { params }),
  getById: (id) => request.get(`/building/${id}`),
  create:   (data) => request.post('/building', data),
  update:   (id, data) => request.put(`/building/${id}`, data),
  delete:   (id) => request.delete(`/building/${id}`)
}
```

### 3.3 Pinia 用户仓库 `src/store/user.js`

```js
import { defineStore } from 'pinia'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    userInfo: null
  }),
  actions: {
    setToken(token) {
      this.token = token
      localStorage.setItem('token', token)
    },
    logout() {
      this.token = ''
      this.userInfo = null
      localStorage.removeItem('token')
    }
  }
})
```

### 3.4 通用 CrudTable 组件 `src/components/CrudTable.vue`

```vue
<template>
  <div class="crud-table">
    <!-- 搜索栏 -->
    <SearchForm :fields="searchFields" @search="handleSearch" />

    <!-- 操作按钮 -->
    <el-button type="primary" @click="$emit('create')">新增</el-button>

    <!-- 数据表格 -->
    <el-table :data="data" border stripe v-loading="loading">
      <el-table-column
        v-for="col in columns"
        :key="col.prop"
        :prop="col.prop"
        :label="col.label"
        :width="col.width"
      />
      <el-table-column label="操作" width="200">
        <template #default="{ row }">
          <el-button size="small" @click="$emit('edit', row)">编辑</el-button>
          <el-button size="small" type="danger" @click="$emit('delete', row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <!-- 分页 -->
    <el-pagination
      v-model:current-page="currentPage"
      v-model:page-size="pageSize"
      :total="total"
      @change="$emit('pageChange', { page: currentPage, pageSize })"
    />
  </div>
</template>

<script setup>
import { ref } from 'vue'

defineProps({
  columns:      { type: Array, required: true },
  data:         { type: Array, default: () => [] },
  total:        { type: Number, default: 0 },
  loading:      { type: Boolean, default: false },
  searchFields: { type: Array, default: () => [] }
})

defineEmits(['create', 'edit', 'delete', 'search', 'pageChange'])

const currentPage = ref(1)
const pageSize = ref(10)
</script>
```

## 五、常见报错

| 报错信息 | 原因 | 解决方案 |
|---|---|---|
| `Network Error` | 后端未启动或端口不对 | 确认后端监听 `5000` 端口，检查 `vite.config.js` 代理配置 |
| `401 Unauthorized` | Token 过期或缺失 | 检查 `localStorage` 中 `token` 是否存在 |
| `Cannot read properties of undefined` | API 返回数据格式不匹配 | 确认后端返回 `{ code, message, data }` 格式 |
| `AxiosError: timeout` | 请求超时 | 检查数据库连接，确认 Oracle 容器正常运行 |
| `Failed to resolve component CrudTable` | 组件未注册 | 在 `main.js` 中全局注册或在页面中局部引入 |
| `Module not found: @/utils/request` | 路径别名未配置 | 在 `vite.config.js` 中配置 `resolve.alias` |
