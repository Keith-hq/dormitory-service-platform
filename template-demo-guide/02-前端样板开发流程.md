## 三、代码模板

### 3.0 开启本地 Mock 登录

后端 JWT 接口未就绪时，在 `.env.local` 中配置：

```bash
VITE_USE_MOCK=true
```

登录页仍统一调用 `authApi.login()`，Mock 只在请求适配层接管 `POST /api/auth/login`。关闭开关后请求会直接进入真实后端，不需要修改页面代码。

临时请求体：

```json
{
  "loginName": "student001",
  "password": "123456"
}
```

临时响应仍遵循统一结构：

```json
{
  "code": 200,
  "message": "登录成功",
  "data": {
    "token": "mock-token",
    "userInfo": {
      "id": "student001",
      "name": "测试学生",
      "role": "student"
    }
  }
}
```

### 3.1 请求封装 `src/utils/request.js`

<!-- prettier-ignore -->
```js
import axios from 'axios'
import { attachAuthMock } from '@/mock/auth'
import { useUserStore } from '@/store/user'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000
})

const AUTH_TOKEN_SNAPSHOT = 'authTokenSnapshot'
let loginRedirectPending = false

const redirectToLogin = (requestToken) => {
  const userStore = useUserStore()
  if (requestToken !== userStore.token) return

  userStore.logout()
  if (window.location.pathname === '/login' || loginRedirectPending) return

  loginRedirectPending = true
  const currentPath = `${window.location.pathname}${window.location.search}${window.location.hash}`
  window.location.assign(`/login?redirect=${encodeURIComponent(currentPath)}`)
}

// 请求拦截器：记录请求使用的 Token、携带 JWT，并按开关接入登录 Mock。
request.interceptors.request.use((config) => {
  const userStore = useUserStore()
  config[AUTH_TOKEN_SNAPSHOT] = userStore.token
  if (userStore.token) {
    config.headers.Authorization = `Bearer ${userStore.token}`
  }
  return attachAuthMock(config)
})

// 业务 401 与 HTTP 401 都必须走同一个会话清理入口。
request.interceptors.response.use(
  (response) => {
    const { code, message, data } = response.data
    if (code === 200 || code === 201) return data
    if (code === 401) {
      redirectToLogin(response.config[AUTH_TOKEN_SNAPSHOT])
    }
    return Promise.reject(new Error(message || '请求失败'))
  },
  (error) => {
    if (error.response?.status === 401) {
      redirectToLogin(error.config?.[AUTH_TOKEN_SNAPSHOT])
    }
    return Promise.reject(error)
  }
)

export default request
```

### 3.3 Pinia 用户仓库 `src/store/user.js`

<!-- prettier-ignore -->
```js
import { defineStore } from 'pinia'

const TOKEN_KEY = 'token'
const USER_INFO_KEY = 'userInfo'

const getLocalStorage = () => {
  try {
    return window.localStorage
  } catch {
    return null
  }
}

const safeGetItem = (key) => {
  try {
    return getLocalStorage()?.getItem(key) ?? null
  } catch {
    return null
  }
}

const safeSetItem = (key, value) => {
  try {
    const storage = getLocalStorage()
    if (!storage) return false
    storage.setItem(key, value)
    return true
  } catch {
    return false
  }
}

const safeRemoveItem = (key) => {
  try {
    getLocalStorage()?.removeItem(key)
  } catch {
    // 存储不可用时退化为 Pinia 内存会话。
  }
}

const clearStoredSession = () => {
  safeRemoveItem(TOKEN_KEY)
  safeRemoveItem(USER_INFO_KEY)
}

const isValidUserInfo = (userInfo) =>
  Boolean(
    userInfo &&
    typeof userInfo === 'object' &&
    !Array.isArray(userInfo) &&
    ['id', 'name', 'role'].every(
      (key) => typeof userInfo[key] === 'string' && userInfo[key].trim()
    )
  )

const readStoredSession = () => {
  const token = safeGetItem(TOKEN_KEY)
  const storedUserInfo = safeGetItem(USER_INFO_KEY)

  try {
    const userInfo = JSON.parse(storedUserInfo)
    if (!token || !isValidUserInfo(userInfo))
      throw new TypeError('登录缓存无效')
    return { token, userInfo }
  } catch {
    clearStoredSession()
    return { token: '', userInfo: null }
  }
}

export const useUserStore = defineStore('user', {
  state: () => readStoredSession(),
  getters: {
    isLoggedIn: (state) => Boolean(state.token && state.userInfo)
  },
  actions: {
    setSession({ token, userInfo }) {
      if (
        typeof token !== 'string' ||
        !token.trim() ||
        !isValidUserInfo(userInfo)
      ) {
        throw new Error('登录响应缺少 Token 或用户信息')
      }

      this.token = token
      this.userInfo = userInfo

      const tokenStored = safeSetItem(TOKEN_KEY, token)
      const userInfoStored = safeSetItem(
        USER_INFO_KEY,
        JSON.stringify(userInfo)
      )
      if (!tokenStored || !userInfoStored) clearStoredSession()
    },
    logout() {
      this.token = ''
      this.userInfo = null
      clearStoredSession()
    }
  }
})
```

### 3.5 路由配置 `src/router/index.js`

<!-- prettier-ignore -->
```js
import { createRouter, createWebHistory } from 'vue-router'
import { getSafeAuthRedirect } from '@/router/authRedirect'
import { useUserStore } from '@/store/user'

const routes = [
  {
    path: '/login',
    name: 'Login',
    component: () => import('@/views/LoginView.vue'),
    meta: { public: true, layout: 'auth' }
  },
  {
    path: '/building',
    name: 'Building',
    component: () => import('@/views/BuildingList.vue'),
    meta: { requiresAuth: true }
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

router.beforeEach((to) => {
  const userStore = useUserStore()
  if (to.meta.requiresAuth && !userStore.isLoggedIn) {
    return { name: 'Login', query: { redirect: to.fullPath } }
  }
  if (to.name === 'Login' && userStore.isLoggedIn) {
    return getSafeAuthRedirect(to.query.redirect)
  }
  return true
})

export default router
```

`getSafeAuthRedirect()` 只允许跳回已登记的站内业务路由，并拒绝跨域地址、反斜杠及登录页循环目标。

## 五、常见报错

| 报错信息                                | 原因                   | 解决方案                                                 |
| --------------------------------------- | ---------------------- | -------------------------------------------------------- |
| `Network Error`                         | 后端未启动或端口不对   | 确认后端监听 `5000` 端口，检查 `vite.config.js` 代理配置 |
| `401 Unauthorized`                      | Token 过期或缺失       | 检查 `localStorage` 中 `token` 是否存在                  |
| `Cannot read properties of undefined`   | API 返回数据格式不匹配 | 确认后端返回 `{ code, message, data }` 格式              |
| `AxiosError: timeout`                   | 请求超时               | 检查数据库连接，确认 Oracle 容器正常运行                 |
| `Failed to resolve component CrudTable` | 组件未注册             | 在 `main.js` 中全局注册或在页面中局部引入                |
| `Module not found: @/utils/request`     | 路径别名未配置         | 在 `vite.config.js` 中配置 `resolve.alias`               |
